unix_pair {
	fd0	sock_unix
	fd1	sock_unix
}

sockaddr_un [
	file	sockaddr_un_file
	abs	sockaddr_un_abstract
] [varlen]

sockaddr_un_file {
	family	const[AF_UNIX, int16]
	path	filename
} [packed]

sockaddr_un_abstract {
	family	const[AF_UNIX, int16]
	ind	const[0, int8]
	id	proc[20000, 4, int32]
} [packed]

type sockaddr_storage array[int8, 128]

msghdr_un {
	addr	ptr[in, sockaddr_un, opt]
	addrlen	len[addr, int32]
	vec	ptr[in, array[iovec_in]]
	vlen	len[vec, int32]
	ctrl	ptr[in, array[cmsghdr_un], opt]
	ctrllen	bytesize[ctrl, int32]
	f	const[0, int32]
}

recv_msghdr {
	addr	ptr[out, sockaddr_storage, opt]
	addrlen	len[addr, int32]
	vec	ptr[in, array[iovec_out]]
	vlen	len[vec, int32]
	ctrl	ptr[out, array[int8], opt]
	ctrllen	bytesize[ctrl, int32]
	f	const[0, int32]
}

cmsghdr_un [
	rights	cmsghdr_un_rights
	cred	cmsghdr_un_cred
] [varlen]

cmsghdr_un_rights {
	len	len[parent, int32]
	pad	const[0, int32]
	level	const[SOL_SOCKET, int32]
	type	const[SCM_RIGHTS, int32]
	fds	array[fd]
} [align[8]]

cmsghdr_un_cred {
	len	const[0x1c, int32]
	pad	const[0, int32]
	level	const[SOL_SOCKET, int32]
	type	const[SCM_CREDENTIALS, int32]
	cred	ucred
} [align[8]]

ucred {
	pid	pid
	uid	uid
	gid	gid
}
